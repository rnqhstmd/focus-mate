+    private func photoRect(_ photo: CardPhoto, size: CGSize) -> CGRect {
+        let width = size.width * photo.widthFraction

+    }
+    private func handleRect(_ photo: CardPhoto, size: CGSize) -> CGRect {
+        let rect = photoRect(photo, size: size)
+        return CGRect(x: rect.maxX - 38, y: rect.maxY - 38, width: 44, height: 44)
+    }

+                style.photos[index].widthFraction = min(2.5, max(0.15, start.widthFraction + gestureTranslation.width / size.width))
+            } else {
+                style.photos[index].centerX = min(2, max(-1, start.centerX + gestureTranslation.width / size.width))
+                style.photos[index].centerY = min(2, max(-1, start.centerY + gestureTranslation.height / size.height))
+            }
+        }
+        return style
+    }

+import SwiftUI
+import CoreText
+
+/// Bundled Seoul city fonts; system text remains the fallback if registration fails.
+enum FontCatalog {
+    static func register() {
+        for name in ["SeoulAlrim-Medium", "SeoulAlrim-Bold"] {
+            if let url = Bundle.main.url(forResource: name, withExtension: "ttf") {
+                CTFontManagerRegisterFontsForURL(url as CFURL, .process, nil)
+            }
+        }
+    }
+
+    static func text(_ size: CGFloat, bold: Bool = false, relativeTo: Font.TextStyle = .body) -> Font {
+        .custom(bold ? "SeoulAlrimTTF-Bold" : "SeoulAlrimTTF-Medium", size: size, relativeTo: relativeTo)
+    }
+}

+/// Only the editor Save action arms this shared box; bootstrap and unrelated saves never arm it.
+final class UITestCardSaveFailure {
+    private let enabled: Bool
+    private var used = false
+    private var armed = false
+    init(enabled: Bool) { self.enabled = enabled }
+    func arm() { if enabled && !used { armed = true } }
+    func consume() -> Bool {
+        guard armed else { return false }
+        armed = false; used = true
+        return true
+    }
+}

+@MainActor
+final class UITestCardImportGate {
+    private var token = UUID()
+    private var waiter: CheckedContinuation<Void, Error>?
+    private var released = false
+    func arm() -> UUID {
+        cancel(token)
+        token = UUID(); released = false
+        return token
+    }
+    func wait(_ expected: UUID) async throws {
+        try Task.checkCancellation()
+        guard expected == token else { throw CancellationError() }
+        if released { return }
+        try await withTaskCancellationHandler {
+            try await withCheckedThrowingContinuation { waiter = $0 }
+        } onCancel: {
+            Task { @MainActor in self.cancel(expected) }
+        }
+    }
+    func release() {
+        released = true
+        let continuation = waiter; waiter = nil
+        continuation?.resume()
+    }
+    private func cancel(_ expected: UUID) {
+        guard expected == token else { return }
+        let continuation = waiter; waiter = nil
+        continuation?.resume(throwing: CancellationError())
+    }

+                       "The fixed decorative card composition must stay identical from large to accessibility5")
+    }

+    private func render(_ textSize: DynamicTypeSize) throws -> CGImage {
+        let content = PassFace(style: .standard)
+            .environment(\.dynamicTypeSize, textSize)
+            .environment(\.colorScheme, .dark)
+            .environment(\.locale, Locale(identifier: "ko_KR"))
+            .frame(width: 352, height: 221.1875)
+        let renderer = ImageRenderer(content: content)
+        renderer.scale = 1
+        renderer.proposedSize = ProposedViewSize(width: 352, height: 221.1875)
+        return try XCTUnwrap(renderer.cgImage, "The actual PassFace must render")
+    }
+
+    private func rgbaPixels(_ image: CGImage) throws -> Data {
+        var bytes = [UInt8](repeating: 0, count: image.width * image.height * 4)
+        let colorSpace = try XCTUnwrap(CGColorSpace(name: CGColorSpace.sRGB))
+        try bytes.withUnsafeMutableBytes { buffer in
+            let context = try XCTUnwrap(CGContext(
+                data: buffer.baseAddress, width: image.width, height: image.height,
+                bitsPerComponent: 8, bytesPerRow: image.width * 4,
+                space: colorSpace,
+                bitmapInfo: CGImageAlphaInfo.premultipliedLast.rawValue
+            ))
+            context.draw(image, in: CGRect(x: 0, y: 0, width: image.width, height: image.height))
+        }
+        return Data(bytes)
+    }
+
+    private func digest(_ pixels: Data) -> String {
+        SHA256.hash(data: pixels).map { String(format: "%02x", $0) }.joined()
+    }

+                          width: viewport.width, height: max(0, bottom - viewport.minY))
+        }
+        func reachable() -> Bool {
+            let fixedIDs = [helperID("close"), helperID("undo"),
+                            "test.cardImportFixture", "test.releaseCardImport"]
+            let viewport: CGRect
+            if target.identifier == helperID("save") {
+                viewport = h.app.frame
+            } else if fixedIDs.contains(target.identifier) {
+                let appFrame = h.app.frame
+                viewport = CGRect(x: appFrame.minX, y: appFrame.minY,
+                                  width: appFrame.width, height: footer.frame.minY - appFrame.minY)
+            } else {
+                viewport = visibleBody()
+            }
+            return target.isHittable && viewport.contains(target.frame)
+        }
+        for _ in 0..<12 {
+            if reachable() { break }
+            let lowerBound = visibleBody().minY
+            let upperBound = visibleBody().maxY
+            if target.frame.maxY > upperBound {
+                try scrollBody(body, contentUp: true)
+            } else if target.frame.minY < lowerBound {
+                try scrollBody(body, contentUp: false)
+            } else {
+                try scrollBody(body, contentUp: target.frame.midY >= body.frame.midY)
+            }
+        }
+        XCTAssertTrue(reachable(), "본문/조작 도달 실패: " + target.identifier)
+    }
+    private func scrollBody(_ body: XCUIElement, contentUp: Bool) throws {
+        XCTAssertTrue(body.waitForExistence(timeout: timeout))
+        let start = body.coordinate(withNormalizedOffset: CGVector(dx: 0.02, dy: contentUp ? 0.8 : 0.2))
+        let end = body.coordinate(withNormalizedOffset: CGVector(dx: 0.02, dy: contentUp ? 0.2 : 0.8))
+        start.press(forDuration: 0.01, thenDragTo: end)
+    }

+    private func successfulSave(_ h: FocusMateUIHarness, expected: Style) throws -> StyleState {
