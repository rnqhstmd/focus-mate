+import Foundation
+import Observation
+
+enum LaunchState: Equatable { case loading, ready(LaunchRoute), failed(StorageIssue) }
+enum Destination: String, Hashable, CaseIterable {
+    case preparation, records, settings, profile
+    var title: String {
+        switch self {
+        case .preparation: "여행 준비"
+        case .records: "기록"
+        case .settings: "설정"
+        case .profile: "마이페이지"
+        }
+    }
+}
+
+@MainActor @Observable
+final class AppModel {
+    private(set) var launchState: LaunchState = .loading
+    private(set) var snapshot: AppSnapshot?
+    var path: [Destination] = []
+    private(set) var saveFailure: StorageIssue?
+    private let store: LocalSnapshotStore
+    @ObservationIgnored private var hasLoaded = false
+    init(store: LocalSnapshotStore) { self.store = store }
+
+    func loadIfNeeded() {
+        guard !hasLoaded else { return }
+        hasLoaded = true
+        do {
+            let restored = try store.load() ?? .empty
+            let route = try restored.launchRoute()
+            snapshot = restored
+            launchState = .ready(route)
+        } catch let issue as StorageIssue {
+            snapshot = nil; path = []; launchState = .failed(issue)
+        } catch {
+            snapshot = nil; path = []; launchState = .failed(.unreadable)
+        }
+    }
+
+    @discardableResult
+    func save(_ candidate: AppSnapshot) -> Bool {
+        guard case let .ready(previousRoute) = launchState else {
+            if case let .failed(issue) = launchState { saveFailure = .writeBlocked(issue) }
+            else { saveFailure = .notReady }
+            return false
+        }
+        do {
+            let nextRoute = try candidate.launchRoute()
+            try store.save(candidate)
+            snapshot = candidate; launchState = .ready(nextRoute); saveFailure = nil
+            if previousRoute != nextRoute { path = [] }
+            return true
+        } catch let issue as SnapshotValidationIssue { saveFailure = .invalidState(issue) }
+        catch let issue as StorageIssue { saveFailure = issue }
+        catch { saveFailure = .writeFailed }
+        return false
+    }
+
+    func open(_ destination: Destination) {
+        guard launchState == .ready(.main) else { return }
+        path.append(destination)
+    }
+    func goBack() {
+        guard launchState == .ready(.main), !path.isEmpty else { return }
+        path.removeLast()
+    }
+}

+import Foundation
+
+struct StationSnapshot: Codable, Equatable {
+    let id: String
+    let name: String
+    let direction: Int
+}
+
+enum LaunchRoute: Equatable { case cardEntry, initialStationEntry, main }
+enum SnapshotValidationIssue: Error, Equatable {
+    case inconsistentStage, emptyStationID, emptyStationName, invalidDirection(Int)
+}
+
+struct AppSnapshot: Codable, Equatable {
+    let cardIssued: Bool
+    let initialStation: StationSnapshot?
+    let currentStation: StationSnapshot?
+    static let empty = AppSnapshot(cardIssued: false, initialStation: nil, currentStation: nil)
+
+    func validate() throws {
+        let noStations = initialStation == nil && currentStation == nil
+        let bothStations = initialStation != nil && currentStation != nil
+        guard noStations || (cardIssued && bothStations) else { throw SnapshotValidationIssue.inconsistentStage }
+        for station in [initialStation, currentStation].compactMap({ $0 }) {
+            guard !station.id.trimmingCharacters(in: .whitespacesAndNewlines).isEmpty else { throw SnapshotValidationIssue.emptyStationID }
+            guard !station.name.trimmingCharacters(in: .whitespacesAndNewlines).isEmpty else { throw SnapshotValidationIssue.emptyStationName }
+            guard station.direction == 1 || station.direction == -1 else { throw SnapshotValidationIssue.invalidDirection(station.direction) }
+        }
+    }
+
+    func launchRoute() throws -> LaunchRoute {
+        try validate()
+        if !cardIssued { return .cardEntry }
+        return currentStation == nil ? .initialStationEntry : .main
+    }
+
+    private enum CodingKeys: String, CodingKey { case cardIssued, initialStation, currentStation }
+    func encode(to encoder: Encoder) throws {
+        var c = encoder.container(keyedBy: CodingKeys.self)
+        try c.encode(cardIssued, forKey: .cardIssued)
+        if let initialStation { try c.encode(initialStation, forKey: .initialStation) }
+        else { try c.encodeNil(forKey: .initialStation) }
+        if let currentStation { try c.encode(currentStation, forKey: .currentStation) }
+        else { try c.encodeNil(forKey: .currentStation) }
+    }
+}

+import Foundation
+
+indirect enum StorageIssue: Error, Equatable {
+    case unreadable, corrupt, unsupportedVersion(Int), invalidState(SnapshotValidationIssue)
+    case writeFailed, writeBlocked(StorageIssue), notReady
+}
+
+final class LocalSnapshotStore {
+    let fileURL: URL
+    private let readData: (URL) throws -> Data
+    private let createDirectory: (URL) throws -> Void
+    private let writeData: (Data, URL) throws -> Void
+
+    init(fileURL: URL,
+         readData: @escaping (URL) throws -> Data = { try Data(contentsOf: $0) },
+         createDirectory: @escaping (URL) throws -> Void = { try FileManager.default.createDirectory(at: $0, withIntermediateDirectories: true) },
+         writeData: @escaping (Data, URL) throws -> Void = { try $0.write(to: $1, options: .atomic) }) {
+        self.fileURL = fileURL
+        self.readData = readData
+        self.createDirectory = createDirectory
+        self.writeData = writeData
+    }
+
+    private struct VersionHeader: Decodable { let version: Int }
+    private struct Envelope: Codable { let version: Int; let state: AppSnapshot }
+
+    func load() throws -> AppSnapshot? {
+        let data: Data
+        do { data = try readData(fileURL) }
+        catch let error as CocoaError where error.code == .fileReadNoSuchFile { return nil }
+        catch { throw StorageIssue.unreadable }
+        let header: VersionHeader
+        do { header = try JSONDecoder().decode(VersionHeader.self, from: data) }
+        catch { throw StorageIssue.corrupt }
+        guard header.version == 1 else { throw StorageIssue.unsupportedVersion(header.version) }
+        do {
+            let state = try JSONDecoder().decode(Envelope.self, from: data).state
+            try state.validate()
+            return state
+        } catch let issue as SnapshotValidationIssue { throw StorageIssue.invalidState(issue) }
+        catch { throw StorageIssue.corrupt }
+    }
+
+    func save(_ snapshot: AppSnapshot) throws {
+        do { try snapshot.validate() }
+        catch let issue as SnapshotValidationIssue { throw StorageIssue.invalidState(issue) }
+        do { _ = try load() }
+        catch let issue as StorageIssue { throw StorageIssue.writeBlocked(issue) }
+        do {
+            let encoder = JSONEncoder()
+            encoder.outputFormatting = [.sortedKeys]
+            let data = try encoder.encode(Envelope(version: 1, state: snapshot))
+            try createDirectory(fileURL.deletingLastPathComponent())
+            try writeData(data, fileURL)
+        } catch { throw StorageIssue.writeFailed }
+    }
+}

+#!/usr/bin/env python3
+"""Build/test the local iOS app, selecting an installed iPhone simulator."""
+import json
+import os
+from pathlib import Path
+import subprocess
+import sys
+
+root = Path(__file__).resolve().parent.parent
+if len(sys.argv) < 2 or sys.argv[1] not in {"build", "test"}:
+    raise SystemExit("Usage: python3 scripts/check-ios.py build|test [test files...]")
+action = sys.argv[1]
+device_id = os.environ.get("FOCUSMATE_SIMULATOR_ID")
+if not device_id:
+    inventory = json.loads(subprocess.check_output(["xcrun", "simctl", "list", "devices", "available", "-j"]))
+    candidates = []
+    for runtime, devices in inventory["devices"].items():
+        if ".iOS-" not in runtime:
+            continue
+        version = tuple(int(part) for part in runtime.split(".iOS-")[1].split("-"))
+        if version[0] < 17:
+            continue
+        for device in devices:
+            if device.get("isAvailable") and device["name"].startswith("iPhone"):
+                candidates.append((version, "Pro" in device["name"] and "Max" not in device["name"], device["name"], device["udid"]))
+    if not candidates:
+        raise SystemExit("No available iPhone simulator with iOS 17 or later. Install a runtime in Xcode.")
+    device_id = max(candidates)[3]
+command = ["xcodebuild", "-project", "ios/FocusMate.xcodeproj", "-scheme", "FocusMate",
+           "-destination", f"platform=iOS Simulator,id={device_id}",
+           "-derivedDataPath", os.environ.get("FOCUSMATE_DERIVED_DATA", str(root / ".build/ios")),
+           "-parallel-testing-enabled", "NO", action, "CODE_SIGNING_ALLOWED=NO"]
+if action == "test":
+    command.extend(["-collect-test-diagnostics", "never"])
+    for file in sys.argv[2:]:
+        path = Path(file)
+        if path.parent.name not in {"FocusMateTests", "FocusMateUITests"} or path.suffix != ".swift":
+            raise SystemExit(f"Unsupported test file: {file}")
+        command.append(f"-only-testing:{path.parent.name}/{path.stem}")
+elif len(sys.argv) != 2:
+    raise SystemExit("Build does not accept test-file arguments.")
+print(f"iOS {action}: simulator {device_id}", flush=True)
+raise SystemExit(subprocess.call(command, cwd=root))
