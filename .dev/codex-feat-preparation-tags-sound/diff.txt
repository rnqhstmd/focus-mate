+@MainActor final class AVSoundPreviewPlayer: SoundPreviewPlaying {
+    private enum ResourceIssue: Error { case missingResource }
+    private let player: AVAudioPlayer
+    var volume: Float {
+        get { player.volume }
+        set { player.volume = newValue }
+    }
+    var numberOfLoops: Int {
+        get { player.numberOfLoops }
+        set { player.numberOfLoops = newValue }
+    }
+    init(url: URL) throws { player = try AVAudioPlayer(contentsOf: url) }
+    func prepareToPlay() -> Bool { player.prepareToPlay() }
+    func play() -> Bool { player.play() }
+    func setVolume(_ volume: Float, fadeDuration: TimeInterval) {
+        player.setVolume(volume, fadeDuration: fadeDuration)
+    }
+    func stop() { player.stop() }
+    static func make(sound: JourneySound, bundle: Bundle = .main) async throws -> any SoundPreviewPlaying {
+        guard let name = sound.resourceName,
+              let url = bundle.url(forResource: name, withExtension: "wav") else {
+            throw ResourceIssue.missingResource
+        }
+        return try AVSoundPreviewPlayer(url: url)
+    }

+struct PreparationDraft: Equatable {
+    private(set) var tags = PreparationTag.defaults
+    private(set) var selectedTagID: String?
+    private(set) var task = ""
+    private(set) var sound: JourneySound = .bus
+    var selectedTag: PreparationTag? { tags.first { $0.id == selectedTagID } }
+    var trimmedTask: String { task.trimmingCharacters(in: .whitespacesAndNewlines) }
+    var isComplete: Bool { selectedTag != nil && !trimmedTask.isEmpty }
+    @discardableResult mutating func chooseTag(id: String) -> Bool {
+        guard tags.contains(where: { $0.id == id }) else { return false }
+        selectedTagID = id
+        return true
+    }
+    mutating func setTask(_ value: String) {
+        var result = String.UnicodeScalarView()
+        var units = 0
+        for scalar in value.unicodeScalars {
+            let width = scalar.value > 0xFFFF ? 2 : 1
+            guard units + width <= 80 else { break }
+            result.append(scalar)
+            units += width
+        }
+        task = String(result)
+    }
+    mutating func setSound(_ value: JourneySound) { sound = value }
+    @discardableResult mutating func saveTag(_ editor: TagEditorDraft) -> Result<PreparationTag, TagNameIssue> {
+        let name = editor.name.trimmingCharacters(in: .whitespacesAndNewlines)
+        guard !name.isEmpty else { return .failure(.empty) }
+        guard name.utf16.count <= 20 else { return .failure(.tooLong) }
+        let normalized = name.precomposedStringWithCompatibilityMapping.lowercased()
+        guard !tags.contains(where: {

+        }) else { return .failure(.duplicate) }

+                                 name: name,

+        tags.append(tag)
+        selectedTagID = tag.id
+        return .success(tag)
+    }

+import Foundation
+import Observation
+
+@MainActor protocol SoundPreviewPlaying: AnyObject {
+    var volume: Float { get set }
+    var numberOfLoops: Int { get set }
+    func prepareToPlay() -> Bool
+    func play() -> Bool
+    func setVolume(_ volume: Float, fadeDuration: TimeInterval)
+    func stop()
+}
+
+@MainActor @Observable final class SoundPreviewController {
+    enum Phase: Equatable { case stopped, loading, playing }
+    private(set) var selectedSound: JourneySound
+    private(set) var phase: Phase = .stopped
+    private(set) var errorMessage: String?
+    private let makePlayer: @MainActor (JourneySound) async throws -> any SoundPreviewPlaying
+    private let sleep: @MainActor (Duration) async -> Void
+    private var player: (any SoundPreviewPlaying)?
+    private var stopTask: Task<Void, Never>?
+    private var generation = 0
+    private var active = true
+    init(sound: JourneySound,
+         makePlayer: @escaping @MainActor (JourneySound) async throws -> any SoundPreviewPlaying,
+         sleep: @escaping @MainActor (Duration) async -> Void = { try? await Task.sleep(for: $0) }) {
+        selectedSound = sound
+        self.makePlayer = makePlayer
+        self.sleep = sleep
+    }
+    func chooseSound(_ sound: JourneySound) {
+        stopPreview()
+        selectedSound = sound
+        errorMessage = nil
+    }
+    func togglePreview() async {
+        guard phase == .stopped else { stopPreview(); return }
+        guard active, selectedSound != .none, !Task.isCancelled else { return }
+        generation += 1
+        let token = generation
+        let sound = selectedSound
+        phase = .loading
+        errorMessage = nil
+        do {
+            let candidate = try await makePlayer(sound)
+            guard isCurrent(token, sound: sound) else {
+                candidate.stop()
+                if generation == token { phase = .stopped }
+                return
+            }
+            if let stopTask { await stopTask.value }
+            guard isCurrent(token, sound: sound) else {
+                candidate.stop()
+                if generation == token { phase = .stopped }
+                return
+            }
+            candidate.volume = 0
+            candidate.numberOfLoops = -1
+            guard candidate.prepareToPlay(), candidate.play() else {
+                candidate.stop()
+                phase = .stopped
+                errorMessage = "소리를 재생하지 못했어요. 선택은 그대로 저장할 수 있어요."
+                return
+            }
+            candidate.setVolume(1, fadeDuration: 0.12)
+            player = candidate
+            phase = .playing
+        } catch {
+            guard isCurrent(token, sound: sound) else {
+                if generation == token { phase = .stopped }
+                return
+            }
+            phase = .stopped
+            errorMessage = "소리를 재생하지 못했어요. 선택은 그대로 저장할 수 있어요."
+        }
+    }
+    func stopPreview() {
+        generation += 1
+        phase = .stopped
+        guard let player else { return }
+        self.player = nil
+        player.setVolume(0, fadeDuration: 0.06)
+        let sleep = self.sleep
+        stopTask = Task {
+            await sleep(.milliseconds(60))
+            player.stop()
+        }
+    }
+    func setActive(_ active: Bool) {
+        self.active = active
+        if !active { stopPreview() }
+    }
+    func beginSelection(sound: JourneySound) {
+        stopPreview()
+        active = true
+        selectedSound = sound
+        phase = .stopped
+        errorMessage = nil
+    }
+
+    private func isCurrent(_ token: Int, sound: JourneySound) -> Bool {
+        token == generation && active && selectedSound == sound && !Task.isCancelled
+    }
+}

+@MainActor private final class PreviewEventLog { var events: [String] = [] }
+private enum PreviewFailure: Error { case unavailable }
+@MainActor private final class PreviewPlayer: SoundPreviewPlaying {
+    let name: String; let log: PreviewEventLog; let prepareResult: Bool; let playResult: Bool
+    let stopped = XCTestExpectation(description: "player stopped")
+    private(set) var playCalls = 0; private(set) var stopCalls = 0
+    var playObservation: XCTestExpectation?
+    var volume: Float = 1 { didSet { log.events.append("\(name).volume=\(volume)") } }
+    var numberOfLoops: Int = 0 { didSet { log.events.append("\(name).loops=\(numberOfLoops)") } }
+    init(_ name: String, log: PreviewEventLog, prepare: Bool = true, play: Bool = true) {
+        self.name = name; self.log = log; prepareResult = prepare; playResult = play
+    }
+    func prepareToPlay() -> Bool { log.events.append("\(name).prepare"); return prepareResult }
+    func play() -> Bool {
+        playCalls += 1; log.events.append("\(name).play"); playObservation?.fulfill()
+        return playResult
+    }
+    func setVolume(_ volume: Float, fadeDuration: TimeInterval) { log.events.append("\(name).fade=\(volume):\(fadeDuration)") }
+    func stop() { stopCalls += 1; log.events.append("\(name).stop"); if stopCalls == 1 { stopped.fulfill() } }
+}
+@MainActor private final class PreviewFactoryGate {
+    private(set) var sounds: [JourneySound] = []
+    private var pending: [Int: CheckedContinuation<any SoundPreviewPlaying, Error>] = [:]
+    private var arrivals: [Int: XCTestExpectation] = [:]
+    func arrival(_ index: Int) -> XCTestExpectation {
+        if let e = arrivals[index] { return e }
+        let e = XCTestExpectation(description: "factory request \(index)"); arrivals[index] = e
+        if sounds.count > index { e.fulfill() }
+        return e
+    }
+    func make(_ sound: JourneySound) async throws -> any SoundPreviewPlaying {
+        let index = sounds.count; let entered = arrival(index); sounds.append(sound)
+        return try await withCheckedThrowingContinuation { c in
+            pending[index] = c; entered.fulfill()
+        }
+    }
+    func resolve(_ index: Int, _ result: Result<any SoundPreviewPlaying, Error>) {
+        guard let c = pending.removeValue(forKey: index) else { XCTFail("factory continuation absent: \(index)"); return }
+        c.resume(with: result)
+    }
+}
+@MainActor private final class PreviewSleepGate {
+    private(set) var durations: [Duration] = []
+    var arrival = XCTestExpectation(description: "fade sleep entered")
+    private var pending: [CheckedContinuation<Void, Never>] = []
+    func sleep(_ duration: Duration) async {
+        durations.append(duration)
+        await withCheckedContinuation { c in pending.append(c); arrival.fulfill() }
+    }
+    func releaseAll() {
+        let continuations = pending; pending.removeAll()
+        arrival = XCTestExpectation(description: "next fade sleep entered")
+        continuations.forEach { $0.resume() }
+    }
