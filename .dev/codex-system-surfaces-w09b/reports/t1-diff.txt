
# ios/Shared/RecordsAnalyticsTypes.swift
import Foundation

enum AnalyticsPeriodKind: String, Codable, Equatable, CaseIterable { case week, month }
enum AnalyticsTimeBand: String, Codable, Equatable, CaseIterable { case dawn, morning, afternoon, evening }
enum AnalyticsLengthBand: String, Codable, Equatable, CaseIterable { case upTo30, from31To60, from61To120, from121To240, from241To360 }
enum AnalyticsTagID: Codable, Equatable, Hashable { case named(String), remainder }

struct AnalyticsPeriod: Codable, Equatable {
    let kind: AnalyticsPeriodKind
    let startDay: String
    let endDay: String
}

struct AnalyticsRecordInput: Codable, Equatable {
    let id: String
    let day: String
    let completedAt: Date
    let minutes: Int
    let tag: String
    let tagColor: String
    let bus: String
    let from: String
    let to: String
}

struct AnalyticsTotals: Codable, Equatable {
    let minutes: Int
    let count: Int
    let activeDays: Int
    let averageMinutes: Int
}

struct AnalyticsTagTotal: Codable, Equatable {
    let id: AnalyticsTagID
    let name: String
    let color: String
    let minutes: Int
    let count: Int
    let share: Double
}

struct AnalyticsDayTotal: Codable, Equatable {
    let day: String
    let minutes: Int
    let count: Int
    let isFuture: Bool
    let tags: [AnalyticsTagTotal]
}

struct AnalyticsComparison: Codable, Equatable {
    let previousStartDay: String?
    let previousEndDay: String?
    let currentMinutes: Int
    let previousMinutes: Int
    let deltaMinutes: Int
    let changePercent: Double?
}

struct AnalyticsTimeBandTotal: Codable, Equatable {
    let band: AnalyticsTimeBand
    let minutes: Int
    let count: Int
}

struct AnalyticsLengthTotal: Codable, Equatable {
    let band: AnalyticsLengthBand
    let minutes: Int
    let count: Int
}

struct AnalyticsRouteTotal: Codable, Equatable {
    let bus: String
    let from: String
    let to: String
    let minutes: Int
    let count: Int
}

struct AnalyticsInsights: Codable, Equatable {
    let timeBands: [AnalyticsTimeBandTotal]
    let needsMoreTimeBandRecords: Bool
    let routes: [AnalyticsRouteTotal]
    let lengths: [AnalyticsLengthTotal]
    let longestStreak: Int
}

struct RecordsAnalyticsReport: Codable, Equatable {
    let period: AnalyticsPeriod
    let today: String
    let totals: AnalyticsTotals
    let days: [AnalyticsDayTotal]
    let tags: [AnalyticsTagTotal]
    let topTags: [AnalyticsTagTotal]
    let comparison: AnalyticsComparison
}

struct RecordsAnalyticsComputation: Equatable {
    let report: RecordsAnalyticsReport
    let effectiveTag: String?
    let filteredDays: [AnalyticsDayTotal]
    let scopedRecordIDs: [String]
    let insights: AnalyticsInsights
}

enum AnalyticsIssue: Error, Equatable { case invalidCalendar, invalidDay(String), invalidPeriod, invalidInput, arithmeticOverflow, notImplemented }

# ios/Shared/RecordsAnalytics.swift
import Foundation

/// 앱·위젯 공용 순수 분석. 저장된 날짜와 표시 시간대를 분리한다.
enum RecordsAnalytics {
    private static var utc: Calendar {
        var value = Calendar(identifier: .gregorian)
        value.timeZone = TimeZone(secondsFromGMT: 0)!
        return value
    }

    private static func check(_ calendar: Calendar) throws {
        guard calendar.identifier == .gregorian else { throw AnalyticsIssue.invalidCalendar }
    }

    private static func key(_ date: Date, in calendar: Calendar = utc) throws -> String {
        let c = calendar.dateComponents([.era, .year, .month, .day], from: date)
        guard c.era == 1, let y = c.year, (1...9999).contains(y), let m = c.month, let d = c.day else {
            throw AnalyticsIssue.invalidInput
        }
        return String(format: "%04d-%02d-%02d", y, m, d)
    }

    private static func date(_ day: String) throws -> Date {
        guard day.utf8.count == 10, day.range(of: "^[0-9]{4}-[0-9]{2}-[0-9]{2}$", options: .regularExpression) != nil else {
            throw AnalyticsIssue.invalidDay(day)
        }
        let parts = day.split(separator: "-").compactMap { Int($0) }
        guard parts.count == 3, (1...9999).contains(parts[0]),
              let result = utc.date(from: DateComponents(era: 1, year: parts[0], month: parts[1], day: parts[2])),
              (try? key(result)) == day else { throw AnalyticsIssue.invalidDay(day) }
        return result
    }

    static func today(now: Date, calendar: Calendar, timeZone: TimeZone) throws -> String {
        try check(calendar)
        guard now.timeIntervalSinceReferenceDate.isFinite else { throw AnalyticsIssue.invalidInput }
        var local = utc
        local.timeZone = timeZone
        return try key(now, in: local)
    }

    static func period(kind: AnalyticsPeriodKind, containing day: String, calendar: Calendar) throws -> AnalyticsPeriod {
        try check(calendar)
        let value = try date(day)
        let start: Date
        let end: Date
        switch kind {
        case .week:
            let back = (utc.component(.weekday, from: value) + 5) % 7
            guard let first = utc.date(byAdding: .day, value: -back, to: value),
                  let last = utc.date(byAdding: .day, value: 6, to: first) else { throw AnalyticsIssue.invalidPeriod }
            start = first; end = last
        case .month:
            let c = utc.dateComponents([.year, .month], from: value)
            guard let first = utc.date(from: DateComponents(era: 1, year: c.year, month: c.month, day: 1)),
                  let range = utc.range(of: .day, in: .month, for: first),
                  let last = utc.date(byAdding: .day, value: range.count - 1, to: first) else { throw AnalyticsIssue.invalidPeriod }
            start = first; end = last
        }
        return try AnalyticsPeriod(kind: kind, startDay: key(start), endDay: key(end))
    }

    private static func validate(_ value: AnalyticsPeriod, calendar: Calendar) throws {
        guard try period(kind: value.kind, containing: value.startDay, calendar: calendar) == value else { throw AnalyticsIssue.invalidPeriod }
    }

    static func shift(period value: AnalyticsPeriod, by offset: Int, today: String, calendar: Calendar) throws -> AnalyticsPeriod? {
        try check(calendar)
        try validate(value, calendar: calendar)
        _ = try date(today)
        guard (-120_000...120_000).contains(offset) else { return nil }
        let component: Calendar.Component = value.kind == .week ? .day : .month
        let amount = value.kind == .week ? offset * 7 : offset
        guard let moved = utc.date(byAdding: component, value: amount, to: try date(value.startDay)),
              let day = try? key(moved),
              let candidate = try? period(kind: value.kind, containing: day, calendar: calendar),
              candidate.startDay <= today else { return nil }
        return candidate
    }

    private static func sum(_ values: [Int]) throws -> Int {
        try values.reduce(0) { total, value in
            let (next, overflow) = total.addingReportingOverflow(value)
            guard !overflow else { throw AnalyticsIssue.arithmeticOverflow }
            return next
        }
    }

    private static func tags(_ records: [AnalyticsRecordInput]) throws -> [AnalyticsTagTotal] {
        let total = try sum(records.map(\.minutes))
        return try Dictionary(grouping: records, by: \.tag).map { name, items in
            let minutes = try sum(items.map(\.minutes))
            let latest = items.dropFirst().reduce(items[0]) { $1.completedAt > $0.completedAt ? $1 : $0 }
            return AnalyticsTagTotal(id: .named(name), name: name, color: latest.tagColor, minutes: minutes,
                                     count: items.count, share: total == 0 ? 0 : Double(minutes) / Double(total))
        }.sorted {
            if $0.minutes != $1.minutes { return $0.minutes > $1.minutes }
            if $0.count != $1.count { return $0.count > $1.count }
            return $0.name < $1.name
        }
    }

    private static func days(_ records: [AnalyticsRecordInput], period: AnalyticsPeriod, today: String) throws -> [AnalyticsDayTotal] {
        let grouped = Dictionary(grouping: records, by: \.day)
        let last = try date(period.endDay)
        var current = try date(period.startDay)
        var result: [AnalyticsDayTotal] = []
        while current <= last {
            let day = try key(current)
            let items = grouped[day] ?? []
            result.append(try AnalyticsDayTotal(day: day, minutes: sum(items.map(\.minutes)), count: items.count,
                                               isFuture: day > today, tags: tags(items)))
            guard let next = utc.date(byAdding: .day, value: 1, to: current), next > current else { throw AnalyticsIssue.invalidPeriod }
            current = next
        }
        return result
    }

    private static func comparison(_ records: [AnalyticsRecordInput], period: AnalyticsPeriod, today: String, currentMinutes: Int, calendar: Calendar) throws -> AnalyticsComparison {
        guard let previous = try shift(period: period, by: -1, today: today, calendar: calendar) else {
            return AnalyticsComparison(previousStartDay: nil, previousEndDay: nil, currentMinutes: currentMinutes,
                                       previousMinutes: 0, deltaMinutes: currentMinutes, changePercent: nil)
        }
        var previousEnd = previous.endDay
        if period.startDay <= today, today <= period.endDay {
            let elapsed = utc.dateComponents([.day], from: try date(period.startDay), to: try date(today)).day ?? 0
            if let end = utc.date(byAdding: .day, value: elapsed, to: try date(previous.startDay)) {
                previousEnd = min(previousEnd, try key(end))
            }
        }
        let previousMinutes = try sum(records.filter { previous.startDay <= $0.day && $0.day <= previousEnd }.map(\.minutes))
        let delta = currentMinutes - previousMinutes
        return AnalyticsComparison(previousStartDay: previous.startDay, previousEndDay: previousEnd, currentMinutes: currentMinutes,
                                   previousMinutes: previousMinutes, deltaMinutes: delta,
                                   changePercent: previousMinutes == 0 ? nil : Double(delta) / Double(previousMinutes) * 100)
    }

    private static func streak(_ records: [AnalyticsRecordInput]) throws -> Int {
        let ordered = Set(records.map(\.day)).sorted()
        var last: Date?
        var length = 0
        var longest = 0
        for day in ordered {
            let current = try date(day)
            length = last.flatMap { utc.date(byAdding: .day, value: 1, to: $0) } == current ? length + 1 : 1
            longest = max(longest, length)
            last = current
        }
        return longest
    }

    private struct Route: Hashable { let bus, from, to: String }

    static func compute(records: [AnalyticsRecordInput], period: AnalyticsPeriod, selectedTag: String?, now: Date, calendar: Calendar, timeZone: TimeZone) throws -> RecordsAnalyticsComputation {
        try check(calendar)
        try validate(period, calendar: calendar)
        let today = try today(now: now, calendar: calendar, timeZone: timeZone)
        var ids = Set<[UInt16]>()
        for record in records {
            _ = try date(record.day)
            guard !record.id.trimmingCharacters(in: .whitespacesAndNewlines).isEmpty, record.id.utf16.count <= 100,
                  ids.insert(Array(record.id.utf16)).inserted, (1...360).contains(record.minutes),
                  record.completedAt.timeIntervalSinceReferenceDate.isFinite,
                  [record.tag, record.bus, record.from, record.to].allSatisfy({ !$0.trimmingCharacters(in: .whitespacesAndNewlines).isEmpty && $0.utf16.count <= 500 }),
                  record.tagColor.range(of: "^#[0-9a-fA-F]{6}$", options: .regularExpression) != nil else { throw AnalyticsIssue.invalidInput }
        }
        let eligible = records.filter { $0.day <= today && $0.completedAt <= now }
        let current = eligible.filter { period.startDay <= $0.day && $0.day <= period.endDay }
        let minutes = try sum(current.map(\.minutes))
        let totals = AnalyticsTotals(minutes: minutes, count: current.count, activeDays: Set(current.map(\.day)).count,
                                     averageMinutes: current.isEmpty ? 0 : Int((Double(minutes) / Double(current.count)).rounded()))
        let allTags = try tags(current)
        var topTags = Array(allTags.prefix(3))
        if allTags.count > 3 {
            let remainder = Array(allTags.dropFirst(3))
            let restMinutes = try sum(remainder.map(\.minutes))
            topTags.append(AnalyticsTagTotal(id: .remainder, name: "기타", color: "#BFC6C1", minutes: restMinutes,
                                            count: try sum(remainder.map(\.count)), share: minutes == 0 ? 0 : Double(restMinutes) / Double(minutes)))
        }
        let effectiveTag = selectedTag.flatMap { name in allTags.contains { $0.name == name } ? name : nil }
        let scoped = current.filter { effectiveTag == nil || $0.tag == effectiveTag }
        let orderedIDs = scoped.enumerated().sorted {
            $0.element.completedAt == $1.element.completedAt ? $0.offset < $1.offset : $0.element.completedAt > $1.element.completedAt
        }.map(\.element.id)
        var local = utc
        local.timeZone = timeZone
        func timeBand(_ item: AnalyticsRecordInput) -> AnalyticsTimeBand {
            switch local.component(.hour, from: item.completedAt) {
            case 0..<5: .dawn
            case 5..<12: .morning
            case 12..<18: .afternoon
            default: .evening
            }
        }
        let timeBands = try AnalyticsTimeBand.allCases.map { band in
            let items = scoped.filter { timeBand($0) == band }
            return try AnalyticsTimeBandTotal(band: band, minutes: sum(items.map(\.minutes)), count: items.count)
        }
        func lengthBand(_ item: AnalyticsRecordInput) -> AnalyticsLengthBand {
            switch item.minutes {
            case 1...30: .upTo30
            case 31...60: .from31To60
            case 61...120: .from61To120
            case 121...240: .from121To240
            default: .from241To360
            }
        }
        let lengths = try AnalyticsLengthBand.allCases.map { band in
            let items = scoped.filter { lengthBand($0) == band }
            return try AnalyticsLengthTotal(band: band, minutes: sum(items.map(\.minutes)), count: items.count)
        }
        let routes = try Dictionary(grouping: scoped, by: { Route(bus: $0.bus, from: $0.from, to: $0.to) }).map { route, items in
            try AnalyticsRouteTotal(bus: route.bus, from: route.from, to: route.to, minutes: sum(items.map(\.minutes)), count: items.count)
        }.sorted {
            if $0.count != $1.count { return $0.count > $1.count }
            if $0.minutes != $1.minutes { return $0.minutes > $1.minutes }
            return ($0.bus, $0.from, $0.to) < ($1.bus, $1.from, $1.to)
        }
        let report = try RecordsAnalyticsReport(period: period, today: today, totals: totals, days: days(current, period: period, today: today),
                                               tags: allTags, topTags: topTags,
                                               comparison: comparison(eligible, period: period, today: today, currentMinutes: minutes, calendar: calendar))
        return try RecordsAnalyticsComputation(report: report, effectiveTag: effectiveTag, filteredDays: days(scoped, period: period, today: today),
                                               scopedRecordIDs: orderedIDs,
                                               insights: AnalyticsInsights(timeBands: timeBands, needsMoreTimeBandRecords: scoped.count < 3,
                                                                           routes: Array(routes.prefix(3)), lengths: lengths, longestStreak: streak(current)))
    }
}

import Foundation
import XCTest
@testable import FocusMate

final class RecordsAnalyticsTests: XCTestCase {
    private var utc: TimeZone { TimeZone(secondsFromGMT: 0)! }
    private var calendar: Calendar {
        var value = Calendar(identifier: .gregorian)
        value.timeZone = utc
        return value
    }
    private func date(_ text: String) throws -> Date {
        try XCTUnwrap(ISO8601DateFormatter().date(from: text))
    }
    private func input(
        _ id: String, day: String = "2026-10-14", minutes: Int = 10,
        tag: String = "A", color: String = "#112233",
        completed: String = "2026-10-14T12:00:00Z",
        bus: String = "2014", from: String = "성수", to: String = "서울숲"
    ) throws -> AnalyticsRecordInput {
        AnalyticsRecordInput(
            id: id, day: day, completedAt: try date(completed), minutes: minutes,
            tag: tag, tagColor: color, bus: bus, from: from, to: to
        )
    }
    private var week: AnalyticsPeriod {
        AnalyticsPeriod(kind: .week, startDay: "2026-10-12", endDay: "2026-10-18")
    }
    private func compute(
        _ records: [AnalyticsRecordInput], period: AnalyticsPeriod? = nil,
        tag: String? = nil, instant: String = "2026-10-16T23:00:00Z",
        zone: TimeZone? = nil
    ) throws -> RecordsAnalyticsComputation {
        try RecordsAnalytics.compute(
            records: records, period: period ?? week, selectedTag: tag,
            now: date(instant), calendar: calendar, timeZone: zone ?? utc
        )
    }

    // 깨짐: 주를 일요일부터 계산하거나 7일을 벗어난다.
    func testW09b_선택수요일의_주는_월요일부터일요일이다() throws {
        let result = try RecordsAnalytics.period(kind: .week, containing: "2026-10-14", calendar: calendar)
        XCTAssertEqual(result.startDay, "2026-10-12")
        XCTAssertEqual(result.endDay, "2026-10-18")
        let report = try compute([], period: result).report
        XCTAssertEqual(report.days.count, 7)
        XCTAssertEqual(report.days.map(\.isFuture), [false, false, false, false, false, true, true])
    }

    // 깨짐: 월 길이를 고정하거나 윤년2월을28일로 처리한다.
    func testW09b_윤년과평년의_2월은29일과28일이다() throws {
        let leap = try RecordsAnalytics.period(kind: .month, containing: "2024-02-14", calendar: calendar)
        let normal = try RecordsAnalytics.period(kind: .month, containing: "2025-02-14", calendar: calendar)
        XCTAssertEqual(leap.endDay, "2024-02-29")
        XCTAssertEqual(normal.endDay, "2025-02-28")
        XCTAssertEqual(try compute([], period: leap).report.days.count, 29)
        XCTAssertEqual(try compute([], period: normal).report.days.count, 28)
    }

    // 깨짐: 연도경계를 넘는 주의 종료일을 같은 해로 만든다.
    func testW09b_연말주는_다음해일요일까지_포함한다() throws {
        let result = try RecordsAnalytics.period(kind: .week, containing: "2026-12-31", calendar: calendar)
        XCTAssertEqual(result.startDay, "2026-12-28")
        XCTAssertEqual(result.endDay, "2027-01-03")
    }

    // 깨짐: DST의23시간날을86400초로 순회하여 날짜를 중복하거나 누락한다.
    func testW09b_DST전환주도_서로다른7일을_반환한다() throws {
        var local = Calendar(identifier: .gregorian)
        let zone = try XCTUnwrap(TimeZone(identifier: "America/New_York"))
        local.timeZone = zone
        let period = try RecordsAnalytics.period(kind: .week, containing: "2026-03-08", calendar: local)
        let result = try RecordsAnalytics.compute(
            records: [], period: period, selectedTag: nil,
            now: date("2026-03-09T12:00:00Z"), calendar: local, timeZone: zone
        )
        XCTAssertEqual(result.report.days.map(\.day), ["2026-03-02", "2026-03-03", "2026-03-04", "2026-03-05", "2026-03-06", "2026-03-07", "2026-03-08"])

    // 깨짐: 주입시간대를 무시하고 UTC날짜를 오늘로 쓴다.
    func testW09b_오늘은_주입서울시간대에서_다음날이다() throws {
        let result = try RecordsAnalytics.today(
            now: date("2026-10-16T23:00:00Z"), calendar: calendar,
            timeZone: XCTUnwrap(TimeZone(identifier: "Asia/Seoul"))
        )
        XCTAssertEqual(result, "2026-10-17")
    }

    // 깨짐: 오늘보다 미래에 시작하는 다음주로 이동을 허용한다.
    func testW09b_미래주이동은_차단하고_이전주는_허용한다() throws {
        XCTAssertNil(try RecordsAnalytics.shift(period: week, by: 1, today: "2026-10-16", calendar: calendar))
        let previous = try XCTUnwrap(RecordsAnalytics.shift(period: week, by: -1, today: "2026-10-16", calendar: calendar))
        XCTAssertEqual(previous.startDay, "2026-10-05")
        XCTAssertEqual(previous.endDay, "2026-10-11")
    }

    // 깨짐: 완료시각을 재분류해 저장day대신 다른 날짜에 합산한다.
    func testW09b_저장day가_완료시각의_표시날짜보다_우선한다() throws {
        let record = try input("saved", day: "2026-10-12", completed: "2026-10-13T23:30:00Z")
        let result = try compute([record], zone: XCTUnwrap(TimeZone(identifier: "Asia/Seoul")))
        XCTAssertEqual(result.report.days.map(\.minutes), [10, 0, 0, 0, 0, 0, 0])
    }

    // 깨짐: 저장day가 미래이거나 completedAt이 미래인 기록을 포함한다.
    func testW09b_미래day와_미래완료시각은_각각_제외한다() throws {
        let records = try [
            input("ok", day: "2026-10-16", minutes: 10, completed: "2026-10-16T22:00:00Z"),
            input("future-day", day: "2026-10-17", minutes: 20, completed: "2026-10-16T22:00:00Z"),
            input("future-time", day: "2026-10-16", minutes: 30, completed: "2026-10-16T23:00:01Z")
        ]
        let result = try compute(records)
        XCTAssertEqual(result.report.totals.minutes, 10)
        XCTAssertEqual(result.scopedRecordIDs, ["ok"])
        let future = try XCTUnwrap(result.report.days.first { $0.day == "2026-10-17" })
        XCTAssertEqual(future.minutes, 0)
        XCTAssertTrue(future.isFuture)
    }

    // 깨짐: 빈날을 빼거나 같은날횟수/집중일수/평균을 혼동한다.
    func testW09b_60분3회2일의_주일별합계와_평균은_일치한다() throws {
        let records = try [input("a", day: "2026-10-12", minutes: 10), input("b", day: "2026-10-12", minutes: 20), input("c", minutes: 30)]
        let report = try compute(records).report
        XCTAssertEqual(report.totals, AnalyticsTotals(minutes: 60, count: 3, activeDays: 2, averageMinutes: 20))
        XCTAssertEqual(report.days.map(\.minutes), [30, 0, 30, 0, 0, 0, 0])
        XCTAssertEqual(report.days.map(\.count), [2, 0, 1, 0, 0, 0, 0])
        let month = AnalyticsPeriod(kind: .month, startDay: "2026-10-01", endDay: "2026-10-31")
        let monthReport = try compute(records, period: month).report
        XCTAssertEqual(monthReport.days.count, 31)
        XCTAssertEqual(monthReport.totals.minutes, 60)
        XCTAssertEqual(monthReport.days.map(\.minutes).reduce(0, +), 60)
    }

    // 깨짐: 평균을 버림하여10.5분을10분으로 만든다.
    func testW09b_평균10점5분은11분으로_반올림한다() throws {
        let result = try compute([input("a", minutes: 10), input("b", minutes: 11)])
        XCTAssertEqual(result.report.totals.averageMinutes, 11)
    }

    // 깨짐: 현재주비교에 이전주전체를 쓰거나 증감을 반대로 계산한다.
    func testW09b_현재주60분은_이전경과5일40분보다20분_증가다() throws {
        let records = try [input("a", minutes: 60), input("old", day: "2026-10-05", minutes: 40), input("excluded", day: "2026-10-10", minutes: 100)]
        let c = try compute(records).report.comparison
        XCTAssertEqual(c.previousStartDay, "2026-10-05")
        XCTAssertEqual(c.previousEndDay, "2026-10-09")
        XCTAssertEqual(c.currentMinutes, 60)
        XCTAssertEqual(c.previousMinutes, 40)
        XCTAssertEqual(c.deltaMinutes, 20)
        XCTAssertEqual(try XCTUnwrap(c.changePercent), 50, accuracy: 0.000001)
    }

    // 깨짐: 이전0분의 비율을 무한대나0퍼센트로 꾸민다.
    func testW09b_이전0분은_비교율없음과_현재증감을_제공한다() throws {
        let c = try compute([input("a", minutes: 30)]).report.comparison
        XCTAssertEqual(c.previousMinutes, 0)
        XCTAssertEqual(c.deltaMinutes, 30)

    // 깨짐: 앱전용원본ID·할일·메모가 위젯소비report JSON으로 유출된다.
    func testW09b_report직렬화에는_ID_할일_메모키가_없다() throws {
        let report = try compute([input("private-record-id", tag: "독서")]).report
        let data = try JSONEncoder().encode(report)
        let json = try JSONSerialization.jsonObject(with: data)
        func keys(_ object: Any) -> [String] {
            if let map = object as? [String: Any] { return Array(map.keys) + map.values.flatMap(keys) }
            if let list = object as? [Any] { return list.flatMap(keys) }
            return []
        }
        let names = Set(keys(json).map { $0.lowercased() })
        XCTAssertTrue(names.isDisjoint(with: ["recordid", "recordids", "task", "memo", "scopedrecordids"]))
        // 태그의 named/remainder 식별자 id는 승인 DTO이며 원본record id와 구분한다.
        XCTAssertFalse(String(decoding: data, as: UTF8.self).contains("private-record-id"))
    }
