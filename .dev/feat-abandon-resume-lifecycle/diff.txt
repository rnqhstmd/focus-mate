+    }

+    }

+final class LifecycleProbe {
+    enum Failure: Error { case expected }
+
+    private(set) var attempts = 0
+    private(set) var writes: [Data] = []
+    var shouldFail = false
+
+    func record(_ data: Data) throws {
+        attempts += 1
+        if shouldFail { throw Failure.expected }
+        writes.append(data)
+    }
