+/// 승차 태그의 순수 기하 규칙. 화면 좌표(카드 기준 위치·태그 영역)만 받아 겹침·끌기 범위·터치 판정을 정한다.
+struct BoardingGeometry {
+    var tagAreaSize = CGSize(width: 238, height: 160)
+    /// 겹친 면적 ÷ 태그 영역 면적이 이 값 이상이면 접촉이다(BR-7).
+    var threshold: CGFloat = 0.30
+    var maxHorizontal: CGFloat = 26
+    var maxDown: CGFloat = 20
+    /// 이 거리 미만으로 움직이고 놓으면 끌기가 아니라 터치다.
+    var tapSlop: CGFloat = 8
+    /// 낭독기·키보드 활성화 위치가 임계를 확실히 넘도록 더하는 여유 높이.
+    var activationMargin: CGFloat = 8
+    /// 시안 kxUUR 기준 승차 화면의 카드 폭.
+    var cardWidth: CGFloat = 268
+
+    /// 좌우는 ±maxHorizontal, 위로는 카드 위쪽이 태그 영역 위쪽에 닿을 때까지, 아래로는 maxDown까지만 움직인다.
+    func clampedOffset(_ translation: CGSize, cardRest: CGRect, tagArea: CGRect) -> CGSize {
+        CGSize(width: min(max(translation.width, -maxHorizontal), maxHorizontal),
+               height: min(max(translation.height, tagArea.minY - cardRest.minY), maxDown))
+    }
+
+    func overlapRatio(card: CGRect, tagArea: CGRect) -> CGFloat {
+        let overlap = card.intersection(tagArea)
+        guard !overlap.isNull, tagArea.width > 0, tagArea.height > 0 else { return 0 }
+        return (overlap.width * overlap.height) / (tagArea.width * tagArea.height)
+    }
+
+    func isContact(card: CGRect, tagArea: CGRect) -> Bool { overlapRatio(card: card, tagArea: tagArea) >= threshold }
+
+    func isTap(_ translation: CGSize) -> Bool { hypot(translation.width, translation.height) < tapSlop }
+
+    /// 카드가 태그 영역 아래쪽 가장자리에 걸치면서 겹침이 임계를 넘는 위치. 카드 폭이 달라도 임계를 넘도록 높이를 계산한다.
+    func activationOffset(cardRest: CGRect, tagArea: CGRect) -> CGSize {
+        let overlapWidth = min(cardRest.width, tagArea.width)
+        guard overlapWidth > 0 else { return .zero }
+        let overlapHeight = threshold * tagArea.width * tagArea.height / overlapWidth + activationMargin
+        return CGSize(width: tagArea.midX - cardRest.midX, height: tagArea.maxY - overlapHeight - cardRest.minY)
+    }
+}

+        case .held, .fading: phase = .abandoned
+        case .dragging:

+            phase = .ready
+        default: break
+        }

+        switch phase {
+        case .abandoned: resetToReady()
+        case .contacting where transitionCancelled: resetWhenSettled = true
+        default: break

+        phase = .ready
+        cardOffset = .zero
+        showsTapHint = false
+        showsSaveFailure = false
+    }

+        transitionToken += 1
+        transitionCancelled = false
+        resetWhenSettled = false
+        showsSaveFailure = false
+        phase = .contacting
+        // 발급과 달리 승차는 동작 줄이기에서 진동도 생략한다(PRD FR-16).
+        if !reduceMotion, haptics.isSupported { haptics.play(.boardingContact) }
+        let id = recognitionID
+        let contact = timing.contact(reduceMotion: reduceMotion)
+        if contact == .zero {
+            commit(id, reduceMotion: reduceMotion)
+        } else {
+            pendingSave = Task { [weak self] in
+                guard let self else { return }
+                await self.sleep(contact)
+                self.commit(id, reduceMotion: reduceMotion)
+            }
+        }
+    }
+
+    private func commit(_ id: Int, reduceMotion: Bool) {
+        guard recognitionID == id, phase == .contacting else { return }
+        guard boarder.boardJourney() else {
+            phase = .failed
+            cardOffset = .zero
+            showsSaveFailure = true
+            announcer.announce(Self.failedAnnouncement)
+            return
+        }
+        announcer.announce(Self.recognizedAnnouncement)
+        if transitionCancelled {
+            if resetWhenSettled { resetToReady() } else { phase = .abandoned }
+            return
+        }
+        phase = .held
+        let token = transitionToken
+        let fade = timing.fade(reduceMotion: reduceMotion)
+        pendingTransition = Task { [weak self] in
+            guard let self else { return }
+            await (self.holdSleep ?? self.sleep)(self.timing.hold)
+            guard self.transitionToken == token, self.phase == .held else { return }
+            self.phase = .fading
+            if fade > .zero { await self.sleep(fade) }
+            guard self.transitionToken == token, self.phase == .fading else { return }
+            self.phase = .finished
+            self.onFinished()
+        }
+    }
+}

+        guard confirmer.confirmDeparture() else {

+            return false
+        }
+        showsSaveFailure = false
+        isConfirmed = true
+        return true
+    }

+        let elapsedSeconds: Int

+    }

+    let id: String
+    let cardID: String
+    let originStopID: String

+    let minutes: Int

+    static func boarding(details: PreparedJourneyDetails, cardID: String, id: String, boardedAt: Date) -> JourneySnapshot {
+        let origin = details.journey.origin
+        let destination = RouteStops.all[RouteStops.wrap(origin.index + origin.direction * details.journey.stops)]
+        return JourneySnapshot(
+            id: id, cardID: cardID, originStopID: RouteStops.all[RouteStops.wrap(origin.index)].id,
+            direction: origin.direction, minutes: details.journey.minutes, busID: details.journey.busID,

+    func validate(currentCardID: String?) throws {
+        let required = [("id", id), ("cardID", cardID), ("originStopID", originStopID), ("busID", busID),

+        // 준비 초안과 같은 규칙: 할 일은 다듬어진 값 그대로·UTF-16 80 이하, 태그 이름은 20 이하.

+    }

+/// sleep 요청마다 대기를 따로 쌓는다. 요청 순서(0부터)로 골라 resume 할 수 있다.
+@MainActor
+final class QueuedSleep {
+    private(set) var requested: [Duration] = []
+    private var continuations: [Int: CheckedContinuation<Void, Never>] = [:]
+    private var waiter: (count: Int, continuation: CheckedContinuation<Void, Never>)?
+
+    func sleep(_ duration: Duration) async {
+        let position = requested.count
+        requested.append(duration)
+        await withCheckedContinuation { (c: CheckedContinuation<Void, Never>) in
+            continuations[position] = c
+            if let w = waiter, requested.count >= w.count {
+                waiter = nil
+                w.continuation.resume()
+            }
+        }
+    }
+
+    /// 요청이 count 건 이상 쌓여 멈출 때까지 기다린다.
+    func untilSuspended(count: Int) async {
+        if requested.count >= count, continuations.count + resumedCount >= count { return }
+        await withCheckedContinuation { (c: CheckedContinuation<Void, Never>) in
+            waiter = (count, c)
+        }
+    }
+
+    private var resumedCount = 0
+
+    func resume(at position: Int) {
+        guard let c = continuations.removeValue(forKey: position) else {
+            XCTFail("resume 할 \(position)번 sleep 이 없다")
+            return
+        }
+        resumedCount += 1
+        c.resume()
+    }
+
+    func resumeOldest() {
+        guard let position = continuations.keys.min() else {
+            XCTFail("resume 할 sleep 이 없다")
+            return
+        }
+        resume(at: position)
+    }
