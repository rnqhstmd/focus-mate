+import Foundation
+
+/// 발급과 기존 사용자 보충이 함께 쓰는 카드 생성기. 난수·시계를 주입해 테스트에서 결정적으로 만든다.
+struct CardSnapshotFactory {
+    private let randomNumber: () -> UInt32
+    private let now: () -> Date
+
+    init(randomNumber: @escaping () -> UInt32 = { .random(in: 0..<100_000_000) },
+         now: @escaping () -> Date = Date.init) {
+        self.randomNumber = randomNumber
+        self.now = now
+    }
+
+    func make() -> CardSnapshot {
+        let digits = String(format: "%08u", randomNumber() % 100_000_000)
+        // 저장 형식이 초 단위라 내려 두어야 저장·로드 왕복 값이 같다.
+        let issuedAt = Date(timeIntervalSince1970: now().timeIntervalSince1970.rounded(.down))
+        return CardSnapshot(id: "FM-" + digits, issuedAt: issuedAt)
+    }
+}

+/// 카드 번호는 그리지 않는다. 식별자(`card.face`)는 호출하는 쪽이 붙인다.

+                Image(Self.dotsAssetName)
+                    .resizable()
+                    .aspectRatio(contentMode: .fit)
+                    .frame(width: proxy.size.width * 0.78)
+                    .opacity(0.85)
+                    .frame(maxWidth: .infinity, maxHeight: .infinity, alignment: .trailing)
+                    .offset(x: proxy.size.width * 0.12)
+                Text("Focus Mate")
+                    .font(.system(size: 10 * scale, weight: .semibold))
+                    .tracking(1 * scale)
+                    .offset(x: 20 * scale, y: 17 * scale)
+                Image(systemName: "wave.3.right")
+                    .resizable()
+                    .scaledToFit()
+                    .frame(width: 22 * scale, height: 22 * scale)
+                    .frame(maxWidth: .infinity, alignment: .trailing)
+                    .padding(.top, 12 * scale)
+                    .padding(.trailing, 16 * scale)

+                    .fill(Color(red: 0xc5 / 255, green: 0xb9 / 255, blue: 0x98 / 255))
+                    .frame(width: proxy.size.width * 0.12, height: proxy.size.height * 0.15)
+                    .offset(x: proxy.size.width * 0.075, y: proxy.size.height * 0.43)
+                Text(PassIdentity.displayName)

+                    .offset(x: 20 * scale, y: proxy.size.height - 44 * scale)

+import Foundation
+
+/// 카드 이름·번호 표기의 단일 출처. 번호 표기는 지금 어떤 화면에도 노출하지 않는다(BR-4).
+struct PassIdentity: Equatable {
+    static let displayName = "집중동행카드"
+
+    let name: String
+    let numberText: String
+
+    init(card: CardSnapshot) {
+        name = Self.displayName
+        let digits = card.id.dropFirst(3)
+        numberText = "FM · \(digits.prefix(4)) \(digits.suffix(4))"
+    }
+
+    static func current(in snapshot: AppSnapshot?) -> PassIdentity? {
+        snapshot?.currentCard.map(PassIdentity.init(card:))
+    }
+}

+import Foundation
+import XCTest
+@testable import FocusMate
+
+final class PassIdentityTests: XCTestCase {
+    private static func snapshot(cardID: String) -> AppSnapshot {
+        let card = CardSnapshot(id: cardID, issuedAt: Date(timeIntervalSince1970: 1_791_882_000))
+        return AppSnapshot.empty.issuingCard(card)
+    }
+
+    // MARK: - AC-15 이름·번호 단일 출처
+
+    // 깨짐: 이름을 displayName 대신 다른 문구로 채우거나 번호 구분자·간격을 바꾸면 실패
+    func testAC15_카드가_있으면_이름과_번호_표기를_돌려준다() {
+        let identity = PassIdentity.current(in: Self.snapshot(cardID: "FM-01234567"))
+
+        XCTAssertEqual(identity?.name, "집중동행카드")
+        XCTAssertEqual(identity?.numberText, "FM · 0123 4567")
+    }
+
+    // 깨짐: 카드가 없을 때 빈 값이나 기본 카드를 만들어 돌려주면 실패
+    func testAC15_카드가_없으면_nil이다() {
+        XCTAssertNil(PassIdentity.current(in: AppSnapshot.empty))
+        XCTAssertNil(PassIdentity.current(in: nil))
+    }
+
+    // 깨짐: displayName 상수를 다른 값으로 바꾸면 실패
+    func testAC15_표시_이름_상수는_집중동행카드다() {
+        XCTAssertEqual(PassIdentity.displayName, "집중동행카드")
+    }
+
+    // 깨짐: 번호를 숫자로 변환했다가 문자열로 되돌려 앞자리 0이 사라지면 실패
+    func testAC15_번호_표기는_앞자리_0을_유지한다() {
+        let identity = PassIdentity(card: CardSnapshot(
+            id: "FM-00000042",
+            issuedAt: Date(timeIntervalSince1970: 1_791_882_000)
+        ))
+
+        XCTAssertEqual(identity.numberText, "FM · 0000 0042")
+    }
+
+    // MARK: - BR-1 카드 ID 팩토리
+
+    // 깨짐: 0 채움 폭이나 FM- 접두를 바꾸면 실패
+    func testBR1_고정_난수는_앞자리_0을_채운_8자리_ID가_된다() {
+        let factory = CardSnapshotFactory(
+            randomNumber: { 1_234_567 },
+            now: { Date(timeIntervalSince1970: 1_791_882_000) }
+        )
+
+        XCTAssertEqual(factory.make().id, "FM-01234567")
+    }
+
+    // 깨짐: % 100_000_000 접기를 빼거나 다른 수로 나누면 실패
+    func testBR1_범위_밖_난수는_나머지로_접는다() {
+        let factory = CardSnapshotFactory(
+            randomNumber: { 123_456_789 },
+            now: { Date(timeIntervalSince1970: 1_791_882_000) }
+        )

+    }
