+    static let standardHold: Duration = .milliseconds(900)
+    /// 화면의 슬롯 연출 길이. hold와 분리해 고정한다 — 테스트가 hold를 붙잡아도 애니메이션은 끝나 앱이 idle이 된다.

+/// 카드 1회 발급 상태 머신. ready/failed에서만 발급을 시작하며, 이 검사가 첫 suspension 전에 MainActor에서 일어나 연타가 직렬화된다.
+@MainActor @Observable
+final class CardIssuanceController {
+    enum Phase: Equatable { case ready, issuing, failed, issued }
+    enum FocusTarget: Equatable { case continueButton }
+
+    static let issuingAnnouncement = "카드를 발급하고 있어요."
+    static let issuedAnnouncement = "카드가 발급됐어요. 출발지를 선택해보세요."
+    static let failedAnnouncement = "카드를 발급하지 못했어요. 다시 시도해 주세요."
+
+    private(set) var phase: Phase = .ready
+    private(set) var focusTarget: FocusTarget?
+
+    @ObservationIgnored private let issuer: CardIssuing
+    @ObservationIgnored private let haptics: IssuanceHaptics
+    @ObservationIgnored private let announcer: AccessibilityAnnouncing
+    @ObservationIgnored private let standardDuration: Duration
+    @ObservationIgnored private let sleep: (Duration) async -> Void
+
+    init(issuer: CardIssuing, haptics: IssuanceHaptics, announcer: AccessibilityAnnouncing,
+         standardDuration: Duration = CardIssuanceTiming.standardHold,
+         sleep: @escaping (Duration) async -> Void = { try? await Task.sleep(for: $0) }) {
+        self.issuer = issuer
+        self.haptics = haptics
+        self.announcer = announcer
+        self.standardDuration = standardDuration
+        self.sleep = sleep
+    }
+
+    static func holdDuration(reduceMotion: Bool, standard: Duration) -> Duration {
+        reduceMotion ? .zero : standard
+    }
+
+    @discardableResult
+    func issue(reduceMotion: Bool) async -> Bool {
+        guard phase == .ready || phase == .failed else { return false }
+        phase = .issuing
+        announcer.announce(Self.issuingAnnouncement)
+        guard issuer.issueCard() else {
+            phase = .failed
+            announcer.announce(Self.failedAnnouncement)
+            return false
+        }
+        // 진동은 연출 시작과 함께, 동작 줄이기와 무관하게 재생한다.
+        if haptics.isSupported { haptics.play(.issuance) }
+        let hold = Self.holdDuration(reduceMotion: reduceMotion, standard: standardDuration)
+        if hold > .zero { await sleep(hold) }
+        phase = .issued
+        announcer.announce(Self.issuedAnnouncement)
+        focusTarget = .continueButton
+        return true
+    }
+
+    func proceed() {
+        guard phase == .issued else { return }
+        issuer.finishCardIssuance()
+    }

+@MainActor
+protocol IssuanceHaptics: AnyObject {
+    var isSupported: Bool { get }

+    func announce(_ message: String)

+/// 지원 판단과 재생을 모두 CoreHaptics로 한다. 엔진·재생 오류는 조용히 무시한다(발급 흐름을 막지 않는다).
+@MainActor
+final class CoreHapticsIssuancePlayer: IssuanceHaptics {
+    let isSupported = CHHapticEngine.capabilitiesForHardware().supportsHaptics
+    private var engine: CHHapticEngine?
+
+    func play(_ pattern: IssuanceHapticPattern) {
+        guard isSupported, let engine = engine ?? makeEngine() else { return }
+        do {
+            try engine.start()
+            let player = try engine.makePlayer(with: pattern.makeCHHapticPattern())
+            try player.start(atTime: CHHapticTimeImmediate)
+        } catch {
+            self.engine = nil
+        }
+    }
+
+    private func makeEngine() -> CHHapticEngine? {
+        guard let engine = try? CHHapticEngine() else { return nil }
+        engine.stoppedHandler = { [weak self] _ in Task { @MainActor in self?.engine = nil } }
+        engine.resetHandler = { [weak self] in Task { @MainActor in self?.engine = nil } }
+        self.engine = engine
+        return engine
+    }
+}
+
+@MainActor
+final class SystemAccessibilityAnnouncer: AccessibilityAnnouncing {
+    func announce(_ message: String) {
+        UIAccessibility.post(notification: .announcement, argument: message)
+    }

+import CoreHaptics
+import Foundation
+
+/// 발급 진동 패턴. 카드가 슬롯에서 나오는 동안 약→강으로 커지는 연속 진동과 끝의 짧은 탭("점점 세게 → 끝에 뚝").
+/// 순수 값으로 두어 테스트가 구조를 직접 확인하고, CoreHaptics 변환은 `makeCHHapticPattern()`이 맡는다.
+struct IssuanceHapticPattern: Equatable {
+    struct CurvePoint: Equatable {
+        let time: TimeInterval
+        let intensity: Float
+    }
+    struct Tap: Equatable {
+        let time: TimeInterval
+        let intensity: Float
+        let sharpness: Float
+    }
+
+    let continuousDuration: TimeInterval
+    let continuousSharpness: Float
+    let intensityCurve: [CurvePoint]
+    let taps: [Tap]
+
+    var duration: TimeInterval { max(continuousDuration, taps.map(\.time).max() ?? 0) }
+
+    static let issuance = IssuanceHapticPattern(
+        continuousDuration: CardIssuanceTiming.hapticDuration,
+        continuousSharpness: 0.3,
+        intensityCurve: [
+            CurvePoint(time: 0, intensity: 0.15),
+            CurvePoint(time: 0.45, intensity: 0.45),
+            CurvePoint(time: CardIssuanceTiming.hapticDuration, intensity: 1.0),
+        ],
+        taps: [Tap(time: CardIssuanceTiming.hapticDuration, intensity: 1.0, sharpness: 0.9)]
+    )
+
+    func makeCHHapticPattern() throws -> CHHapticPattern {
+        let continuous = CHHapticEvent(
+            eventType: .hapticContinuous,
+            parameters: [
+                CHHapticEventParameter(parameterID: .hapticIntensity, value: 1.0),
+                CHHapticEventParameter(parameterID: .hapticSharpness, value: continuousSharpness),
+            ],
+            relativeTime: 0,
+            duration: continuousDuration
+        )
+        let tapEvents = taps.map { tap in
+            CHHapticEvent(
+                eventType: .hapticTransient,
+                parameters: [
+                    CHHapticEventParameter(parameterID: .hapticIntensity, value: tap.intensity),
+                    CHHapticEventParameter(parameterID: .hapticSharpness, value: tap.sharpness),
+                ],
+                relativeTime: tap.time
+            )
+        }
+        let curve = CHHapticParameterCurve(
+            parameterID: .hapticIntensityControl,
+            controlPoints: intensityCurve.map { .init(relativeTime: $0.time, value: $0.intensity) },
+            relativeTime: 0
+        )
+        return try CHHapticPattern(events: [continuous] + tapEvents, parameterCurves: [curve])
+    }
+}

+/// sleep 진입을 기록하고, resume 될 때까지 멈춰 있는다.
+@MainActor
+final class SuspendingSleep {
+    private(set) var requested: [Duration] = []
+    private var continuation: CheckedContinuation<Void, Never>?
+    private var waiter: CheckedContinuation<Void, Never>?
+
+    func sleep(_ duration: Duration) async {
+        requested.append(duration)
+        await withCheckedContinuation { (c: CheckedContinuation<Void, Never>) in
+            continuation = c
+            waiter?.resume()
+            waiter = nil
+        }
+    }
+
+    func untilSuspended() async {
+        if continuation != nil { return }
+        await withCheckedContinuation { (c: CheckedContinuation<Void, Never>) in
+            waiter = c
+        }
+    }
+
+    func resume() {
+        guard let c = continuation else {
+            XCTFail("resume 할 sleep 이 없다")
+            return
+        }
+        continuation = nil
+        c.resume()
+    }
+}
