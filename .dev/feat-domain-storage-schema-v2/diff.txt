+import Foundation
+
+/// 버전 1 저장 계약. 이후 `AppSnapshot`이 바뀌어도 고치지 않는다.
+struct LegacySnapshotV1: Decodable {
+    let cardIssued: Bool
+    let initialStation: StationSnapshot?
+    let currentStation: StationSnapshot?
+}
+
+/// 저장 파일 바이트와 `AppSnapshot` 사이의 변환. 판정 순서는 헤더 → 버전 → 구조 → 내용이다.
+enum SnapshotCodec {
+    static let currentVersion = 2
+
+    private struct VersionHeader: Decodable { let version: Int }
+    private struct EnvelopeV1: Decodable { let state: LegacySnapshotV1 }
+    private struct Envelope: Codable { let version: Int; let state: AppSnapshot }
+
+    static func decode(_ data: Data) throws -> AppSnapshot {
+        let header: VersionHeader
+        do { header = try JSONDecoder().decode(VersionHeader.self, from: data) }
+        catch { throw StorageIssue.corrupt }
+        let state: AppSnapshot
+        do {
+            switch header.version {
+            case 1: state = migrateV1(try JSONDecoder().decode(EnvelopeV1.self, from: data).state)
+            case currentVersion: state = try JSONDecoder().decode(Envelope.self, from: data).state
+            default: throw StorageIssue.unsupportedVersion(header.version)
+            }
+        } catch let issue as StorageIssue { throw issue }
+        catch { throw StorageIssue.corrupt }
+        do { try state.validate() }
+        catch let issue as SnapshotValidationIssue { throw StorageIssue.invalidState(issue) }
+        return state
+    }
+
+    static func encode(_ snapshot: AppSnapshot) throws -> Data {
+        let encoder = JSONEncoder()
+        encoder.outputFormatting = [.sortedKeys]
+        return try encoder.encode(Envelope(version: currentVersion, state: snapshot))
+    }
+
+    static func migrateV1(_ legacy: LegacySnapshotV1) -> AppSnapshot {
+        AppSnapshot.empty.updatingStage(cardIssued: legacy.cardIssued,
+                                        initialStation: legacy.initialStation,
+                                        currentStation: legacy.currentStation)
+    }
+}

+import Foundation

+/// 저장 계층 테스트가 함께 쓰는 헬퍼.

+        AppSnapshot.empty.updatingStage(cardIssued: issued, initialStation: initial, currentStation: current)
+    }
+
+    func assertStorageIssue(
+        _ expected: StorageIssue,
+        _ message: String = "",
+        file: StaticString = #filePath,
+        line: UInt = #line,
+        operation: () throws -> Void
+    ) {
+        do {
+            try operation()
+            XCTFail("Expected StorageIssue \(expected) \(message)", file: file, line: line)
+        } catch let actual as StorageIssue {
+            XCTAssertEqual(actual, expected, message, file: file, line: line)
+        } catch {
+            XCTFail("Expected StorageIssue \(expected) but got \(error) \(message)", file: file, line: line)
+        }
+    }
+
+    func withTemporaryFile(_ operation: (URL) throws -> Void) throws {
+        let root = FileManager.default.temporaryDirectory
+            .appendingPathComponent("FocusMate-\(type(of: self))-\(UUID().uuidString)", isDirectory: true)
+        let fileURL = root
+            .appendingPathComponent("nested", isDirectory: true)
+            .appendingPathComponent("FocusMate", isDirectory: true)
+            .appendingPathComponent("app-state.json", isDirectory: false)
+        defer { try? FileManager.default.removeItem(at: root) }
+        try operation(fileURL)
+    }

+            XCTAssertEqual(try Data(contentsOf: fileURL), firstBytes)
+        }
+    }
+
+    // 깨짐 명명: v1 로드 실패 또는 저장 바이트가 G2와 다르면(키 누락·version 1·정류장 값 손실) 실패.
+    func testAC2b_v1에서_읽은_상태를_저장하면_정확한_v2바이트가_기록된다() throws {
+        try withTemporaryFile { fileURL in
+            try seed(fileURL, Self.v1Seongsu)
+            let store = LocalSnapshotStore(fileURL: fileURL)
+            let loaded = try XCTUnwrap(try store.load())
+
+            try store.save(loaded)
+
+            XCTAssertEqual(try Data(contentsOf: fileURL), Self.g2Seongsu)
+            let reloaded = try XCTUnwrap(try LocalSnapshotStore(fileURL: fileURL).load())
+            XCTAssertEqual(reloaded, Self.stage(true, Self.seongsu, Self.seongsu))
+        }
+    }
+
+    // 깨짐 명명: 미지원/손상 파일을 읽어 버리거나 다른 오류 종류를 던지면 실패.
+    func testAC3_보호대상_파일은_읽기_오류를_던진다() throws {
+        for item in Self.protectedCases {
+            try withTemporaryFile { fileURL in
+                try seed(fileURL, item.bytes)
+                let store = LocalSnapshotStore(fileURL: fileURL)
+
+                assertStorageIssue(item.issue, "load: \(item.name)") {
+                    _ = try store.load()
+                }
+            }
+        }
+    }
+
+    // 깨짐 명명: 보호 파일 위에 덮어쓰거나 디렉토리 생성/쓰기를 시도하면 실패.
+    func testAC3_보호대상_파일은_저장이_차단되고_바이트가_보존된다() throws {
+        for item in Self.protectedCases {
+            try withTemporaryFile { fileURL in
+                try seed(fileURL, item.bytes)
+                var directoryCalls = 0
+                var writeCalls = 0
+                let store = LocalSnapshotStore(
+                    fileURL: fileURL,
+                    createDirectory: { _ in directoryCalls += 1 },
+                    writeData: { _, _ in writeCalls += 1 }
+                )
+
+                assertStorageIssue(.writeBlocked(item.issue), "save: \(item.name)") {
+                    try store.save(Self.stage(true, Self.seongsu, Self.seoulForest))
+                }
+                XCTAssertEqual(directoryCalls, 0, item.name)
+                XCTAssertEqual(writeCalls, 0, item.name)
+                XCTAssertEqual(try Data(contentsOf: fileURL), item.bytes, item.name)
+            }
+        }
+    }
+
+    // 깨짐 명명: 내용 불일치 v1을 통과시키면 load가 값을 반환해 실패.
+    func testAC4a_내용불일치_v1은_읽기_오류이고_저장이_차단된다() throws {
+        try withTemporaryFile { fileURL in
+            try seed(fileURL, Self.v1Inconsistent)
+            var writeCalls = 0
+            let store = LocalSnapshotStore(fileURL: fileURL, writeData: { _, _ in writeCalls += 1 })
+
+            assertStorageIssue(.invalidState(.inconsistentStage)) {
+                _ = try store.load()
+            }
+            assertStorageIssue(.writeBlocked(.invalidState(.inconsistentStage))) {
+                try store.save(Self.stage(true, Self.seongsu, Self.seoulForest))
+            }
+            XCTAssertEqual(writeCalls, 0)
+            XCTAssertEqual(try Data(contentsOf: fileURL), Self.v1Inconsistent)
+        }
+    }
+
+    // 깨짐 명명: 쓰기 오류를 삼키거나 .writeFailed 외 오류로 바꾸거나 기존 파일을 직접 변형하면 실패.
+    func testAC4b_쓰기실패는_writeFailed이고_기존_v2파일이_보존된다() throws {
+        try withTemporaryFile { fileURL in
+            try seed(fileURL, Self.g2Seongsu)
+            let store = LocalSnapshotStore(
+                fileURL: fileURL,
+                writeData: { _, _ in throw ProbeError.expected }
+            )
+
+            assertStorageIssue(.writeFailed) {
+                try store.save(Self.stage(true, Self.seongsu, Self.seoulForest))
+            }
+            XCTAssertEqual(try Data(contentsOf: fileURL), Self.g2Seongsu)
+        }
+    }
+
+    private static let protectedCases: [(name: String, bytes: Data, issue: StorageIssue)] = [
+        ("버전 3", Data(#"{"version":3,"state":null}"#.utf8), .unsupportedVersion(3)),
+        ("버전 999", Data(#"{"version":999,"state":null}"#.utf8), .unsupportedVersion(999)),
+        ("버전 0", Data(#"{"version":0,"state":null}"#.utf8), .unsupportedVersion(0)),
+        ("빈 파일", Data(), .corrupt),
+        ("버전 없음", Data(#"{"state":{}}"#.utf8), .corrupt),
+        ("버전 문자열", Data(#"{"version":"2"}"#.utf8), .corrupt),
+        ("v2 records 누락", v2MissingRecords, .corrupt),
+    ]
+
+    private func seed(_ fileURL: URL, _ bytes: Data) throws {
+        try FileManager.default.createDirectory(
+            at: fileURL.deletingLastPathComponent(),
+            withIntermediateDirectories: true
+        )
+        try bytes.write(to: fileURL)
+    }
