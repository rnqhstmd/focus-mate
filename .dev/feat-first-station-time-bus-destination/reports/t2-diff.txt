+import Foundation
+
+/// 준비 진입점에 넘기는 여정 선택(용어 사전의 "여정 선택"). 저장하지 않는다.
+struct JourneySelection: Equatable {
+    let origin: StopPosition
+    let minutes: Int
+    let busID: String
+    let busNumber: String
+    let destination: String
+    let stops: Int
+}
+
+enum ProceedState: Equatable {
+    case originUnavailable, noCandidates, needsSelection, ready(JourneySelection)
+
+    var title: String {
+        switch self {
+        case .originUnavailable: "출발 정류장을 확인할 수 없어요"
+        case .noCandidates: "이 시간의 도착 후보가 없어요"
+        case .needsSelection: "버스와 도착지를 선택하세요"
+        case .ready: "이 여정 준비하기"
+        }
+    }
+
+    var isEnabled: Bool {
+        if case .ready = self { return true }
+        return false
+    }
+}
+
+/// 시간 선택 화면의 상태. 앱 실행 중 메모리에만 둔다.
+struct JourneyPlanDraft: Equatable {
+    private(set) var minutes: Int = JourneyPlanner.initialMinutes
+    private(set) var selectedBusID: String?
+
+    /// 범위 안으로 맞춘 값이 실제로 바뀌면 같은 버스가 새 시간에도 후보여도 선택을 해제한다.
+    mutating func setMinutes(_ value: Int) {
+        let clamped = JourneyPlanner.clamp(value)
+        guard clamped != minutes else { return }
+        minutes = clamped
+        selectedBusID = nil
+    }
+
+    mutating func choose(busID: String, from origin: StopPosition) -> Bool {
+        guard candidates(from: origin).contains(where: { $0.bus.id == busID }) else { return false }
+        selectedBusID = busID
+        return true
+    }
+
+    func candidates(from origin: StopPosition) -> [JourneyCandidate] {
+        JourneyPlanner.candidates(minutes: minutes, from: origin)
+    }
+
+    func selectedCandidate(from origin: StopPosition) -> JourneyCandidate? {
+        candidates(from: origin).first { $0.bus.id == selectedBusID }
+    }
+
+    func proceedState(from origin: StopPosition) -> ProceedState {
+        if let candidate = selectedCandidate(from: origin) {
+            return .ready(JourneySelection(origin: origin, minutes: minutes, busID: candidate.bus.id,
+                                           busNumber: candidate.bus.number, destination: candidate.destination,
+                                           stops: candidate.stops))
+        }
+        return candidates(from: origin).isEmpty ? .noCandidates : .needsSelection
+    }
+}

+import Foundation
+
+/// 첫 출발지 검색 화면의 상태. 선택은 현재 검색 결과 안에 있을 때만 유효하다.
+struct StationSearchState: Equatable {
+    var query: String = ""
+    private(set) var selectedID: String?
+
+    var results: [StationOption] { StationCatalog.search(query) }
+    var selectedOption: StationOption? { results.first { $0.id == selectedID } }
+    var confirmationText: String? { selectedOption.map { "\($0.name) · \($0.boundLabel)에서 출발" } }
+    var canConfirm: Bool { selectedOption != nil }
+
+    mutating func select(_ optionID: String) -> Bool {
+        guard results.contains(where: { $0.id == optionID }) else { return false }
+        selectedID = optionID
+        return true
+    }
+}

+import Foundation
+import XCTest
+@testable import FocusMate
+
+final class StationSearchStateTests: XCTestCase {
+    private func reverseOptionID(_ state: StationSearchState) throws -> String {
+        try XCTUnwrap(state.results.first { $0.position == StopPosition(index: 0, direction: -1) }).id
+    }
+
+    func test_초기_상태는_3건_결과이고_선택_확정이_없다() {
+        let state = StationSearchState()
+        XCTAssertEqual(state.results.count, 3)
+        XCTAssertNil(state.selectedOption)
+        XCTAssertFalse(state.canConfirm)
+        XCTAssertNil(state.confirmationText)
+    }
+
+    func test_없는역_검색은_결과_0건이다() {
+        var state = StationSearchState()
+        state.query = "없는역"
+        XCTAssertEqual(state.results.count, 0)
+        XCTAssertFalse(state.canConfirm)
+    }
+
+    func test_결과가_없을_때_카탈로그_id_선택은_거부되고_확정할_수_없다() {
+        var state = StationSearchState()
+        state.query = "없는역"
+        for option in StationCatalog.all {
+            XCTAssertFalse(state.select(option.id))
+        }
+        XCTAssertNil(state.selectedOption)
+        XCTAssertFalse(state.canConfirm)
+    }
+
+    func test_성수역_역방향_선택은_성공하고_확인_문구를_만든다() throws {
+        var state = StationSearchState()
+        let id = try reverseOptionID(state)
+        XCTAssertTrue(state.select(id))
+        XCTAssertEqual(state.confirmationText, "성수역 · 성수사거리 방면에서 출발")
+        XCTAssertTrue(state.canConfirm)
+        XCTAssertEqual(state.selectedOption?.position, StopPosition(index: 0, direction: -1))
+    }
+
+    func test_선택한_옵션이_검색_결과에서_빠지면_선택이_사라진다() throws {
+        var state = StationSearchState()
+        XCTAssertTrue(state.select(try reverseOptionID(state)))
+        state.query = "서울숲"
+        XCTAssertNil(state.selectedOption)
+        XCTAssertFalse(state.canConfirm)
+    }
+
+    func test_현재_결과에_없는_옵션_선택은_거부되고_기존_선택을_유지한다() throws {
+        var state = StationSearchState()
+        state.query = "서울숲"
+        let forestID = try XCTUnwrap(state.results.first { $0.position == StopPosition(index: 4, direction: 1) }).id
+        XCTAssertTrue(state.select(forestID))
+        let reverseID = try XCTUnwrap(StationCatalog.all.first { $0.position == StopPosition(index: 0, direction: -1) }).id
+        XCTAssertFalse(state.select(reverseID))
+        XCTAssertEqual(state.selectedOption?.position, StopPosition(index: 4, direction: 1))
+    }
+}
