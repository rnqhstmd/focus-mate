+"""Minimal standard-library xlsx reader for the Seoul bus source files."""
+from dataclasses import dataclass
+import posixpath
+import re
+import xml.etree.ElementTree as ET
+import zipfile
+
+_MAIN = "{http://schemas.openxmlformats.org/spreadsheetml/2006/main}"
+_REL = "{http://schemas.openxmlformats.org/package/2006/relationships}"
+_DOC_REL = "{http://schemas.openxmlformats.org/officeDocument/2006/relationships}"
+_TEXT_TYPES = {"s", "inlineStr", "str"}
+
+
+@dataclass(frozen=True)
+class Cell:
+    value: str
+    is_text: bool
+
+
+@dataclass(frozen=True)
+class Sheet:
+    header: list
+    rows: list
+
+
+class MissingColumnsError(Exception):
+    def __init__(self, missing):
+        self.missing = list(missing)
+        super().__init__("필수 열이 없습니다: " + ", ".join(self.missing))
+
+
+def require_columns(sheet, names):
+    missing = [name for name in names if name not in sheet.header]
+    if missing:
+        raise MissingColumnsError(missing)
+
+
+def read_first_sheet(path):
+    with zipfile.ZipFile(path) as archive:
+        shared = _shared_strings(archive)
+        grid = _sheet_rows(archive, _first_sheet_path(archive), shared)
+    if not grid:
+        return Sheet(header=[], rows=[])
+    header_row = grid[0][1]
+    columns = sorted(header_row)
+    header = [header_row[col].value.strip() for col in columns]
+    names = {col: header_row[col].value.strip() for col in columns}
+    rows = [{names[col]: cell for col, cell in cells.items() if col in names} for _, cells in grid[1:]]
+    return Sheet(header=header, rows=rows)
+
+
+def _first_sheet_path(archive):
+    workbook = ET.fromstring(archive.read("xl/workbook.xml"))
+    sheet = workbook.find(f"{_MAIN}sheets/{_MAIN}sheet")
+    if sheet is None:
+        raise ValueError("시트가 없습니다")
+    rel_id = sheet.get(f"{_DOC_REL}id")
+    rels = ET.fromstring(archive.read("xl/_rels/workbook.xml.rels"))
+    for rel in rels.iter(f"{_REL}Relationship"):
+        if rel.get("Id") == rel_id:
+            target = rel.get("Target")
+            return target.lstrip("/") if target.startswith("/") else posixpath.normpath(posixpath.join("xl", target))
+    raise ValueError("첫 시트 경로를 찾을 수 없습니다")
+
+
+def _text(node):
+    """Joins every <t> under the node except phonetic runs (<rPh>)."""
+    parts = []
+    for child in node:
+        if child.tag == f"{_MAIN}t":
+            parts.append(child.text or "")
+        elif child.tag == f"{_MAIN}r":
+            parts.extend(t.text or "" for t in child.iter(f"{_MAIN}t"))
+    return "".join(parts)
+
+
+def _shared_strings(archive):
+    if "xl/sharedStrings.xml" not in archive.namelist():
+        return []
+    root = ET.fromstring(archive.read("xl/sharedStrings.xml"))
+    return [_text(si) for si in root.iter(f"{_MAIN}si")]
+
+
+def _column_index(reference):
+    letters = re.match(r"[A-Z]+", reference).group()
+    index = 0
+    for letter in letters:
+        index = index * 26 + ord(letter) - ord("A") + 1
+    return index
+
+
+def _cell(element, shared):
+    kind = element.get("t", "n")
+    value = element.find(f"{_MAIN}v")
+    if kind == "inlineStr":
+        inline = element.find(f"{_MAIN}is")
+        text = _text(inline) if inline is not None else ""
+    elif kind == "s":
+        text = shared[int(value.text)] if value is not None else ""
+    else:
+        text = value.text or "" if value is not None else ""
+    return Cell(value=text, is_text=kind in _TEXT_TYPES)
+
+
+def _sheet_rows(archive, sheet_path, shared):
+    rows = []
+    with archive.open(sheet_path) as stream:
+        for _, element in ET.iterparse(stream):
+            if element.tag != f"{_MAIN}row":
+                continue
+            cells = {_column_index(c.get("r")): _cell(c, shared) for c in element.iter(f"{_MAIN}c")}
+            rows.append((int(element.get("r")), cells))
+            element.clear()
+    return rows

+"""테스트용 최소 xlsx 생성 헬퍼 (zipfile만 사용)."""
+import zipfile
+from xml.sax.saxutils import escape
+
+
+class Inline:
+    def __init__(self, text):
+        self.text = text
+
+
+class Num:
+    def __init__(self, raw):
+        self.raw = raw
+
+
+class Bool:
+    def __init__(self, flag):
+        self.flag = flag
+
+
+class Rich:
+    """리치 텍스트 공유 문자열. parts는 <r><t> 조각, phonetic은 <rPh><t> 읽기 기호."""
+
+    def __init__(self, parts, phonetic=None):
+        self.parts = parts
+        self.phonetic = phonetic
+
+
+def _col(i):
+    s = ""
+    i += 1
+    while i:
+        i, r = divmod(i - 1, 26)
+        s = chr(65 + r) + s
+    return s
+
+
+def _t(text):
+    return '<t xml:space="preserve">%s</t>' % escape(text)
+
+
+def write_xlsx(path, rows, *, sheet_file="sheet1.xml"):
+    """rows: 행 목록, 행은 셀 목록. str=공유 문자열, None=생략 칸,
+    Inline/Num/Bool/Rich=해당 셀. sheet_file이 sheet1.xml이 아니면
+    워크북 첫 시트를 그 파일로 매핑하고 sheet1.xml은 미끼 둘째 시트로 둔다."""
+    shared = []
+    sheet_rows = []
+    for rn, row in enumerate(rows, start=1):
+        cells = []
+        for ci, c in enumerate(row):
+            ref = "%s%d" % (_col(ci), rn)
+            if c is None:
+                continue
+            if isinstance(c, str):
+                shared.append("<si>%s</si>" % _t(c))
+                cells.append('<c r="%s" t="s"><v>%d</v></c>' % (ref, len(shared) - 1))
+            elif isinstance(c, Rich):
+                body = "".join("<r>%s</r>" % _t(p) for p in c.parts)
+                if c.phonetic is not None:
+                    body += '<rPh sb="0" eb="1">%s</rPh>' % _t(c.phonetic)
+                shared.append("<si>%s</si>" % body)
+                cells.append('<c r="%s" t="s"><v>%d</v></c>' % (ref, len(shared) - 1))
+            elif isinstance(c, Inline):
+                cells.append('<c r="%s" t="inlineStr"><is>%s</is></c>' % (ref, _t(c.text)))
+            elif isinstance(c, Num):
+                cells.append('<c r="%s"><v>%s</v></c>' % (ref, c.raw))
+            elif isinstance(c, Bool):
+                cells.append('<c r="%s" t="b"><v>%d</v></c>' % (ref, 1 if c.flag else 0))
+            else:
+                raise TypeError(c)
+        sheet_rows.append('<row r="%d">%s</row>' % (rn, "".join(cells)))
+    ns = 'xmlns="http://schemas.openxmlformats.org/spreadsheetml/2006/main"'
+    sheet_xml = '<?xml version="1.0" encoding="UTF-8"?><worksheet %s><sheetData>%s</sheetData></worksheet>' % (
+        ns, "".join(sheet_rows))
+    decoy_xml = ('<?xml version="1.0" encoding="UTF-8"?><worksheet %s><sheetData>'
+                 '<row r="1"><c r="A1" t="inlineStr"><is><t>DECOY</t></is></c></row>'
+                 '</sheetData></worksheet>' % ns)
+    sst = '<?xml version="1.0" encoding="UTF-8"?><sst %s count="%d" uniqueCount="%d">%s</sst>' % (
+        ns, len(shared), len(shared), "".join(shared))
+    rel = "http://schemas.openxmlformats.org/officeDocument/2006/relationships"
+    pkg = "http://schemas.openxmlformats.org/package/2006/relationships"
+    decoy = sheet_file != "sheet1.xml"
+    sheets = '<sheet name="Main" sheetId="1" r:id="rId2"/>' if decoy else '<sheet name="Main" sheetId="1" r:id="rId1"/>'
+    rels = ""
+    if decoy:
+        sheets += '<sheet name="Decoy" sheetId="2" r:id="rId1"/>'
+        rels += '<Relationship Id="rId1" Type="%s/worksheet" Target="worksheets/sheet1.xml"/>' % rel
+        rels += '<Relationship Id="rId2" Type="%s/worksheet" Target="worksheets/%s"/>' % (rel, sheet_file)
+    else:
+        rels += '<Relationship Id="rId1" Type="%s/worksheet" Target="worksheets/sheet1.xml"/>' % rel
+    rels += '<Relationship Id="rId9" Type="%s/sharedStrings" Target="sharedStrings.xml"/>' % rel
+    workbook = ('<?xml version="1.0" encoding="UTF-8"?><workbook %s xmlns:r="%s"><sheets>%s</sheets></workbook>'
+                % (ns, rel, sheets))
+    with zipfile.ZipFile(path, "w", zipfile.ZIP_DEFLATED) as z:
+        z.writestr("[Content_Types].xml",
+                   '<?xml version="1.0" encoding="UTF-8"?><Types xmlns="http://schemas.openxmlformats.org/package/2006/content-types"/>')
+        z.writestr("_rels/.rels",
+                   '<?xml version="1.0" encoding="UTF-8"?><Relationships xmlns="%s"><Relationship Id="rId1" Type="%s/officeDocument" Target="xl/workbook.xml"/></Relationships>' % (pkg, rel))
+        z.writestr("xl/workbook.xml", workbook)
+        z.writestr("xl/_rels/workbook.xml.rels",
+                   '<?xml version="1.0" encoding="UTF-8"?><Relationships xmlns="%s">%s</Relationships>' % (pkg, rels))
+        z.writestr("xl/worksheets/" + sheet_file, sheet_xml)
+        if decoy:
+            z.writestr("xl/worksheets/sheet1.xml", decoy_xml)
+        z.writestr("xl/sharedStrings.xml", sst)
